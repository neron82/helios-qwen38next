cmake_minimum_required(VERSION 3.20)
project(attn_core LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 86)
add_library(attn_core STATIC attn.cu qwen_rope.cu qwen_gqa.cu
  qsa.cu kv_quant.cu)
target_include_directories(attn_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
find_package(CUDAToolkit REQUIRED)
target_link_libraries(attn_core PUBLIC CUDA::cudart)
add_executable(test_attn_parity test/test_attn_parity.cpp)
target_link_libraries(test_attn_parity PRIVATE attn_core)
add_executable(test_qsa_parity test/test_qsa_parity.cpp)
target_link_libraries(test_qsa_parity PRIVATE attn_core aux exl3_quant_core)
add_executable(test_kv_quant test/test_kv_quant.cpp)
target_link_libraries(test_kv_quant PRIVATE attn_core)
